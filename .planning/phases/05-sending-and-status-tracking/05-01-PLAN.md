    ```javascript
    import { addMonths, isAfter, startOfDay, differenceInDays, format } from 'date-fns';

    export function calculateDeadline(sentAt) {
      return addMonths(new Date(sentAt), 1);
    }

    export function isOverdue(deadline) {
      const today = startOfDay(new Date());
      const deadlineDate = startOfDay(new Date(deadline));
      return isAfter(today, deadlineDate);
    }

    export function getDeadlineInfo(deadline) {
      const today = startOfDay(new Date());
      const deadlineDate = startOfDay(new Date(deadline));
      const daysLeft = differenceInDays(deadlineDate, today);

      if (daysLeft < 0) {
        return { label: `${Math.abs(daysLeft)} days overdue`, color: 'danger', overdue: true, daysLeft };
      } else if (daysLeft < 7) {
        return { label: `${daysLeft} days left`, color: 'danger', overdue: false, daysLeft };
      } else if (daysLeft < 14) {
        return { label: `${daysLeft} days left`, color: 'warning', overdue: false, daysLeft };
      } else {
        return { label: `${daysLeft} days left`, color: 'success', overdue: false, daysLeft };
      }
    }

    export function formatDeadlineDate(deadline) {
      return format(new Date(deadline), 'MMM d');
    }
    ```

    ```javascript
    import { calculateDeadline } from './deadline.js';

    /**
     * Update a broker's status with full history tracking.
     * Per D-41: campaign.statuses[brokerId] = { status, sentAt, deadline, history[] }
     * Per D-37: when status is 'awaiting', auto-calculate deadline from sentAt.
     */
    export function updateBrokerStatus(brokerId, status, details = {}) {
      const statuses = { ...campaign.value.statuses };
      const existing = statuses[brokerId] || { history: [] };
      const now = new Date().toISOString();

      const update = { ...existing, status };

      if (status === 'awaiting') {
        update.sentAt = details.sentAt || now;
        update.deadline = details.deadline || calculateDeadline(update.sentAt).toISOString();
      }

      if (details.sentAt) update.sentAt = details.sentAt;
      if (details.deadline) update.deadline = details.deadline;

      update.history = [
        ...(existing.history || []),
        { status, at: now, ...(details.details ? { details: details.details } : {}) },
      ];

      statuses[brokerId] = update;

      campaign.value = {
        ...campaign.value,
        updatedAt: now,
        statuses,
      };
    }

    /**
     * Get aggregate stats for the campaign dashboard.
     * Per D-26: counts for each status category.
     */
    export function getAggregateStats() {
      const statuses = campaign.value.statuses;
      const selected = campaign.value.brokers.selected || [];
      const counts = { total: selected.length, sent: 0, awaiting: 0, confirmed: 0, rejected: 0, overdue: 0, escalated: 0, acknowledged: 0 };

      for (const brokerId of selected) {
        const entry = statuses[brokerId];
        if (!entry) continue;
        const s = entry.status;
        if (s === 'awaiting' || s === 'acknowledged') counts.awaiting++;
        if (s === 'confirmed') counts.confirmed++;
        if (s === 'rejected') counts.rejected++;
        if (s === 'overdue') counts.overdue++;
        if (s === 'escalated') counts.escalated++;
        if (s === 'awaiting' || s === 'acknowledged' || s === 'confirmed' || s === 'rejected' || s === 'overdue' || s === 'escalated') counts.sent++;
      }

      return counts;
    }

    /**
     * Get campaign progress percentage.
     * Per D-25: (confirmed + rejected) / total selected * 100.
     */
    export function getProgressPercentage() {
      const stats = getAggregateStats();
      if (stats.total === 0) return 0;
      return Math.round(((stats.confirmed + stats.rejected) / stats.total) * 100);
    }
    ```

    ```javascript
    import { signal, computed } from '@preact/signals';

    // ---- Batch State Signals ----
    export const batchState = signal({
      active: false,
      cancelled: false,
      total: 0,
      sent: 0,
      failed: [],     // broker IDs that failed after retries
      current: null,   // broker ID currently being sent
      log: [],         // { brokerId, brokerName, status: 'sent'|'failed'|'skipped', at: ISO, error? }
    });

    export const batchProgress = computed(() => {
      const s = batchState.value;
      if (s.total === 0) return 0;
      return Math.round(((s.sent + s.failed.length) / s.total) * 100);
    });

    export function cancelBatch() {
      batchState.value = { ...batchState.value, cancelled: true };
    }

    export function resetBatchState() {
      batchState.value = {
        active: false, cancelled: false,
        total: 0, sent: 0, failed: [], current: null, log: [],
      };
    }

    // ---- EmailJS Sending (per D-01, uses global emailjs object) ----
    export async function sendViaEmailJS(broker, template, config) {
      const { publicKey, serviceId, templateId } = config;
      if (typeof globalThis.emailjs === 'undefined') {
        throw new Error('EmailJS SDK not loaded');
      }
      return globalThis.emailjs.send(serviceId, templateId, {
        to_email: broker.email,
        subject: template.subject,
        body: template.body,
        reply_to: config.tempEmailAddress || '',
        from_name: config.fromName || 'ErasureKit User',
      }, { publicKey });
    }

    // ---- Mailto Fallback (per D-05, SEND-04) ----
    export function generateMailtoLink(brokerEmail, subject, body) {
      const params = new URLSearchParams();
      params.set('subject', subject);
      params.set('body', body);
      const url = `mailto:${brokerEmail}?${params.toString()}`;
      if (url.length > 2000) {
        return { type: 'clipboard', url: null, reason: 'Template exceeds mailto: URL limit (~2000 chars)' };
      }
      return { type: 'mailto', url };
    }

    // ---- Batch Sending Loop (per D-02, D-03, D-04) ----
    function delay(ms) {
      return new Promise(resolve => setTimeout(resolve, ms));
    }

    export async function sendBatch(brokerIds, sendFn, options = {}) {
      const delayMs = options.delayMs ?? 2000;
      const maxRetries = options.maxRetries ?? 3;
      const retryDelay = options.retryDelay ?? 3000;

      batchState.value = {
        active: true, cancelled: false,
        total: brokerIds.length, sent: 0, failed: [], current: null, log: [],
      };

      for (const brokerId of brokerIds) {
        if (batchState.value.cancelled) {
          batchState.value = {
            ...batchState.value,
            log: [...batchState.value.log, { brokerId, status: 'skipped', at: new Date().toISOString() }],
          };
          continue;
        }

        batchState.value = { ...batchState.value, current: brokerId };

        let success = false;
        let lastError = null;
        for (let attempt = 0; attempt < maxRetries && !success; attempt++) {
          try {
            await sendFn(brokerId);
            success = true;
          } catch (err) {
            lastError = err;
            if (attempt < maxRetries - 1) await delay(retryDelay);
          }
        }

        if (success) {
          batchState.value = {
            ...batchState.value,
            sent: batchState.value.sent + 1,
            log: [...batchState.value.log, { brokerId, status: 'sent', at: new Date().toISOString() }],
          };
        } else {
          batchState.value = {
            ...batchState.value,
            failed: [...batchState.value.failed, brokerId],
            log: [...batchState.value.log, { brokerId, status: 'failed', at: new Date().toISOString(), error: lastError?.message }],
          };
        }

        // D-02: 2-3s delay between sends (randomized)
        if (brokerId !== brokerIds[brokerIds.length - 1]) {
          await delay(delayMs + Math.random() * 1000);
        }
      }

      batchState.value = { ...batchState.value, active: false, current: null };
    }

    // ---- Failed Emails Text Download (per D-05) ----
    export function generateFailedEmailsText(failedBrokerIds, brokers, getTemplateFn) {
      const lines = failedBrokerIds.map(id => {
        const broker = brokers.find(b => b.id === id);
        if (!broker) return '';
        const template = getTemplateFn(broker);
        return [
          `--- ${broker.name} ---`,
          `To: ${broker.email}`,
          `Subject: ${template.subject}`,
          '',
          template.body,
          '',
          '---',
          '',
        ].join('\n');
      });
      return lines.join('\n');
    }
    ```
