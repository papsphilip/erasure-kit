    ```javascript
    import { signal, computed } from '@preact/signals';

    // Raw broker list (populated after loading brokers.json)
    export const allBrokers = signal([]);

    // UI filter signals
    export const searchQuery = signal('');
    export const regionFilter = signal('');    // '' = all regions
    export const categoryFilter = signal('');  // '' = all categories
    export const sortField = signal('name');   // 'name' | 'region' | 'category'
    export const sortDirection = signal('asc'); // 'asc' | 'desc'

    // Expand/collapse state (D-03)
    export const expandedBrokerId = signal(null);

    // Derived: filtered + sorted broker list
    export const filteredBrokers = computed(() => {
      let result = allBrokers.value;

      // Search by name (case-insensitive, D-08)
      const query = searchQuery.value.toLowerCase().trim();
      if (query) {
        result = result.filter(b => b.name.toLowerCase().includes(query));
      }

      // Filter by region AND category (D-09)
      if (regionFilter.value) {
        result = result.filter(b => b.region === regionFilter.value);
      }
      if (categoryFilter.value) {
        result = result.filter(b => b.category === categoryFilter.value);
      }

      // Sort (D-01)
      const field = sortField.value;
      const dir = sortDirection.value === 'asc' ? 1 : -1;
      result = [...result].sort((a, b) => {
        const aVal = (a[field] || '').toLowerCase();
        const bVal = (b[field] || '').toLowerCase();
        return aVal < bVal ? -dir : aVal > bVal ? dir : 0;
      });

      return result;
    });

    // Toggle expand/collapse (D-03)
    export function toggleExpand(brokerId) {
      expandedBrokerId.value =
        expandedBrokerId.value === brokerId ? null : brokerId;
    }

    // Toggle sort direction, or switch field (D-01)
    export function toggleSort(field) {
      if (sortField.value === field) {
        sortDirection.value = sortDirection.value === 'asc' ? 'desc' : 'asc';
      } else {
        sortField.value = field;
        sortDirection.value = 'asc';
      }
    }

    // Clear all filters (D-11)
    export function clearAllFilters() {
      searchQuery.value = '';
      regionFilter.value = '';
      categoryFilter.value = '';
    }

    // Initialize: load brokers and validate IDs (Pitfall 5 from research)
    export async function initBrokers() {
      if (allBrokers.value.length > 0) return;
      const { loadBrokers } = await import('./brokers-loader.js');
      const brokers = await loadBrokers();
      allBrokers.value = brokers.map(b => ({
        ...b,
        id: b.id || b.name.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/(^-|-$)/g, ''),
      }));
    }
    ```

    ```javascript
    // ── Broker Selection Helpers (Phase 3) ───────────────────────────────────────

    /**
     * Select a single broker by ID. No-op if already selected.
     * @param {string} brokerId - Broker slug ID
     */
    export function selectBroker(brokerId) {
      const brokers = campaign.value.brokers || {};
      const selected = new Set(brokers.selected || []);
      selected.add(brokerId);
      campaign.value = {
        ...campaign.value,
        updatedAt: new Date().toISOString(),
        brokers: { ...brokers, selected: [...selected] },
      };
    }

    /**
     * Deselect a single broker by ID. No-op if not selected.
     * @param {string} brokerId - Broker slug ID
     */
    export function deselectBroker(brokerId) {
      const brokers = campaign.value.brokers || {};
      const selected = (brokers.selected || []).filter(id => id !== brokerId);
      campaign.value = {
        ...campaign.value,
        updatedAt: new Date().toISOString(),
        brokers: { ...brokers, selected },
      };
    }

    /**
     * Toggle a broker's selection state.
     * @param {string} brokerId - Broker slug ID
     */
    export function toggleBroker(brokerId) {
      const selected = campaign.value.brokers?.selected || [];
      if (selected.includes(brokerId)) {
        deselectBroker(brokerId);
      } else {
        selectBroker(brokerId);
      }
    }

    /**
     * Select multiple brokers at once (bulk select). Deduplicates with existing.
     * Per D-13: operates on provided IDs (UI passes filtered visible broker IDs).
     * @param {string[]} brokerIds - Array of broker slug IDs
     */
    export function selectBrokers(brokerIds) {
      const brokers = campaign.value.brokers || {};
      const selected = new Set([...(brokers.selected || []), ...brokerIds]);
      campaign.value = {
        ...campaign.value,
        updatedAt: new Date().toISOString(),
        brokers: { ...brokers, selected: [...selected] },
      };
    }

    /**
     * Deselect all brokers. Per D-14: "X selected (x)" clears all.
     */
    export function deselectAllBrokers() {
      campaign.value = {
        ...campaign.value,
        updatedAt: new Date().toISOString(),
        brokers: { ...campaign.value.brokers, selected: [] },
      };
    }
    ```
